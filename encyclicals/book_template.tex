% 1. Define your exact desired distance between text and the block
\newlength{\blockspacing}
\setlength{\blockspacing}{8pt} % Alter this to change the space globally

% 2. Create a hidden box to temporarily store and measure your content
\newsavebox{\specialblockbox}

% 3. The non-floating fixed block container
\newenvironment{fixedblock}{%
  \par
  \addvspace{\blockspacing}
  \noindent
  \begin{minipage}{\textwidth}
}{%
  \end{minipage}
  \par
  \addvspace{\blockspacing}
}

% 4. The dynamic wrapper
\newcommand{\placespecialblock}[1]{%
  % A. Save the content inside our measurement box
  \sbox{\specialblockbox}{\begin{minipage}{\textwidth}#1\end{minipage}}%
  %
  % B. Calculate the exact space required: 
  %    Height (\ht) + Depth (\dp) of the block + spacing + 2 lines of text buffer
  \needspace{\dimexpr\ht\specialblockbox + \dp\specialblockbox + \blockspacing + 2\baselineskip\relax}%
  %
  % C. Render the actual block safely
  \begin{fixedblock}
    #1
  \end{fixedblock}%
}

% This macro creates a full-page layout for any chapter you want to feature this way.
\newcommand{\fullpagechapter}[4]{% 1: Title, 2: Subtitle, 3: Author, 4: Year
    \chapter{#1}    
    \thispagestyle{empty} % Removes headers/footers from this special page

    \begin{center}
        
        %--- SUBTITLE ---
        {\Large\itshape #2}
        
        \vfill
        \rule{4cm}{0.4pt} % Clean divider rule
        \vfill
        
        %--- AUTHOR ---
        {\LARGE\scshape #3}

        %--- YEAR ---
        {\small\scshape #4}

        \vfill
    \end{center}
    \newpage % Forces the actual text of the chapter to start on the next page
}
